\chapter{Probabilidade}

\section{Distribuições}

\begin{theorem}[Família exponencial]
	Pertencem à família exponencial:
	\begin{outline}
		\1 \textbf{Discretas}: Bernoulli  ($0<p<1$), Binomial ($n$ fixo, $0<p<1$), Poisson, Binomial Negativa ($0<p<1$), Geométrica;
		\1 \textbf{Contínuas}: Exponencial, Normal, Gama, Qui-quadrado, Beta, Normal Inversa, Weibull.
	\end{outline}
\end{theorem}

\subsection{Distribuições discretas}

\subsubsection{Uniforme discreta(N)}
\begin{outline}
	\1 \( f_X(x \mid N) = \frac{1}{N} \ind_{1, \ldots, N}(x), \quad N \in \N \)
	\1 \( \E[X] = \frac{N+1}{2} \)
	\1 \( \Var[X] = \frac{(N+1)(N-1)}{12} \)
	\1 \( M_X(t) = \frac{1}{N} \sum_{i=1}^{N}e^{it} \) 
\end{outline}

\subsubsection{Bernoulli(p)}
\begin{outline}
	\1 \( f_X(x \mid p) =p^x(1-p)^{1-x} \ind_{\{0,1\}}(x), \quad 0 \leq p \leq 1 \)
	\1 \( \E[X] = p \)
	\1 \( \Var[X] = p(1-p) \)
	\1 \( M_X(t) = (1-p) + pe^t \) 
\end{outline}

\subsubsection{Binomial(n,p)}
\begin{outline}
	\1 \(\displaystyle f_X(x \mid n,p) =\binom{n}{x} p^x(n-p)^{1-x} \ind_{\{0,1,\ldots,n\}}(x), \quad 0 \leq p \leq 1 \)
	\1 \( \E[X] = np \)
	\1 \( \Var[X] = np(1-p) \)
	\1 \( M_X(t) = \left[(1-p) + pe^t\right]^n \) 
\end{outline}

\subsubsection{Poisson($\theta$)}
\begin{outline}
	\1 \(\displaystyle f_X(x \mid \theta) = \frac{\theta^x e^{-\theta}}{x!} \ind_{\{0,1,\ldots\}}(x), \quad \theta \geq 0\)
	\1 \( \E[X] = \theta \)
	\1 \( \Var[X] = \theta \)
	\1 \(\displaystyle M_X(t) = e^{\theta(e^t-1)} \) 
\end{outline}

\subsection{Distribuições contínuas}
\subsubsection{Uniforme(a,b)}
\begin{outline}
	\1 \(\displaystyle f_X(x \mid a,b) = \frac{1}{b-a}\ind_{(a,b)(x)}, \quad a,b \in \Re \)
	\1 \(\displaystyle F_X(x) = \frac{x-a}{b-a}\ind_{(a,b)(x)} + \ind_{[b, \infty](x)}\)
	\1 \( \E[X] = \frac{a+b}{2} \)
	\1 \( \Var[X] = \frac{1}{12}(b-a)^2 \)
	\1 \( M_X(t) = \left\{ 
	\begin{array}{lcl}\displaystyle
		\frac{e^{tb}-e^{ta}}{t(b-a)}, & \text{para} & t\neq0 \\
		1, & \text{para} & t=0
	\end{array} \right. \)
	\1 \(\hat{a}=X_{(1)}, \hat{b}=X_{(n)}\)
\end{outline}

\subsubsection{Exponencial($\lambda$)}
\begin{outline}
	\1 \(\displaystyle f_X(x \mid \lambda) = \lambda e^{-\lambda x} \ind_{\Re_+}(x), \quad \lambda>0\)
	\1 \( F_X(x) = (1-e^{-\lambda x})\ind_{\Re_+}(x) \)
	\1 \( \E[X] = \frac{1}{\lambda} \)
	\1 \( \Var[X] = \frac{1}{\lambda^2} \)
	\1 \(\displaystyle M_X(t) = \frac{\lambda}{\lambda-t}\), para  $t<\lambda$
	\1 \( \hat{\lambda} = 1/\bar{X} \)
\end{outline}

\subsubsection{Normal($\mu, \sigma^2$)}
\begin{outline}
	\1 \(\displaystyle f_X(x\mid \mu, \sigma^2) = \frac{1}{\sqrt{\mu, \sigma^2}}\exp\left\{{-\frac{1}{2\sigma^2}}(x-\mu)^2\right\} \ind_{\Re}(x), \quad \mu\in\Re, \sigma>0\)
	\1 \( \E[X] = \mu \)
	\1 \( \Var[X] = \sigma^2 \)
	\1 \(\displaystyle M_X(t) = e^{\mu t + \sigma^2 t^2/2} \), para  $t<\lambda$
	\1 \( \hat{\mu} = \bar{X}, \hat{\sigma^2} = \left(\frac{n+1}{n}\right) S^2 \)
\end{outline}

\subsubsection{Distribuição t $\tdist{n}$}
\begin{outline}
	\1 Se $Z\sim\normdist{0,1}$ e $U\sim\chidist{n}$ são variáveis aleatórias independentes, então
	\begin{equation}
		T = \frac{Z}{\sqrt{\dfrac{U}{n}}}
	\end{equation}
	tem distribuição t-Student com $n$ graus de liberdade.
	\1 \(\displaystyle f_T(t) = \frac{\Gamma\left(\frac{n+1}{2}\right)}{\Gamma\left(\frac{n}{2}\right)\sqrt{n\pi}}\left(1+\frac{t^2}{n}\right)^{-\left(\frac{n+1}{2}\right)}, \quad -\infty<t<\infty\)
	\1 Não existe função geradora de momentos.
	\1 $\E[T] = 0$, para $n>1$
	\1 $\Var[T] = \dfrac{n}{n-2}$, para $n>2$
\end{outline}

\subsubsection{Distribuição t não central $\tdist{n,\mu}$}
\begin{outline}
	\1 Se $X\sim\normdist{\mu,1}$ e $U\sim\chidist{n}$ são variáveis aleatórias independentes, então
	\begin{equation}
		T^* = \frac{X}{\sqrt{\dfrac{U}{n}}}
	\end{equation}
	tem distribuição t de Student não central, com $n$ graus de liberdade e parâmetro de não centralidade $\mu$.
	\1 \(\displaystyle f_{T^*}(t) = \frac{n^{n/2}e^{-\mu^2/2}}{\Gamma\left(\frac{n}{2}\right)(n+t^2)^{(n+1)/2}}\sum_{k=0}^{\infty}\frac{\Gamma\left(\frac{n+k+1}{2}\right)\mu^k 2^{k/2}t^k}{k!(n+t^2)^{k/2}}\)
\end{outline}

\subsubsection{Qui-quadrado $\chidist{n}$}
\begin{outline}
	\1 \(\displaystyle f_U(u\mid n) = \frac{1}{\Gamma\left(\frac{n}{2}\right)2^{n/2}}u^{(n-2)/2}e^{-u/2}\,\ind_{(0,\infty)}(u), \quad n\geq 1 \text{ inteiro}\)
	\1 \( \E[U] = n \)
	\1 \( \E[U^2] = n(n+2) \)
	\1 \( \Var[U] = 2n \)
	\1 \(\displaystyle M_U(t) = (1-2t)^{-n/2} \), para $t<\frac{1}{2}$
\end{outline}

\begin{theorem}[Qui-quadrado]
	\leavevmode
	\begin{outline}[enumerate]
		\1 Se $Z \sim \normdist{0,1}$, então $Z^2 \sim \chidist{1}$;
		
		\1 Se $X_1, X_2, \ldots, X_n$ são v.a.'s independentes tais que $X_i \sim \chidist{r_i}$, então $\sum_{i=1}^{n} X_i \sim \chidist{r}$, em que $r = \sum_{i=1}^{n} r_i$.
	\end{outline}
\end{theorem}

\subsubsection{Qui-quadrado não central $\chidist{n,\lambda}$}
\begin{outline}
	\1 \(\displaystyle f_U(u\mid n,\lambda) = e^{-\lambda}\sum_{k=0}^{\infty}\frac{\lambda^k}{k!}\frac{u^{(n+2k-2)/2}e^{-u/2}}{\Gamma\left(\frac{n+2k}{2}\right)2^{(n+2k)/2}}\,\ind_{(0,\infty)}(u), \quad n\geq 1 \text{ inteiro}, \ \lambda\geq 0\)
	\1 \( \E[U] = n + 2\lambda \)
	\1 \( \Var[U] = 2(n + 4\lambda) \)
	\1 \(\displaystyle M_U(t) = (1-2t)^{-n/2}\exp\left\{-\lambda\left[1-(1-2t)^{-1}\right]\right\} \), para $t<\frac{1}{2}$
\end{outline}

\subsubsection{Beta($\alpha, \beta$)}
\begin{outline}
	\1 \(\displaystyle f_X(x \mid \alpha, \beta) = \frac{\Gamma(\alpha+\beta)}{\Gamma(\alpha)\Gamma(\beta)} x^{\alpha-1}(1-x)^{\beta-1}\ind_{\Re_{[0,1]}}(x), \quad \alpha,\beta>0 \)
	\1 \(\displaystyle \E[X] = \frac{\alpha}{\alpha+\beta} \)
	\1 \(\displaystyle \Var[X] = \frac{\alpha\beta}{(\alpha+\beta)^2(\alpha+\beta+1)} \)
	\1 \(\displaystyle M_X(t) = 1 + \sum_{k=1}^{\infty} \left( \prod_{r=0}^{k-1} \frac{\alpha+r}{\alpha+\beta+r} \right)\frac{t^k}{k!} \)
\end{outline}

\subsubsection{Gama($\alpha, \theta$) (\textit{shape}, \textit{scale})}
Obs.: parametrização padrão da função rgamma(N, shape, scale) do R.
\begin{outline}
	\1 \(\displaystyle f_X(x \mid \alpha, \beta) = \frac{\beta^\alpha}{\Gamma(\alpha)} x^{\alpha-1}e^{-x/\theta}\ind_{\Re_+}(x), \quad \alpha,\theta>0 \)
	\1 \(\displaystyle \E[X] = \alpha\theta \)
	\1 \(\displaystyle \Var[X] = \alpha\theta^2 \)
	\1 Moda: $(\alpha-1)\theta$, para $\alpha\geq1$; $0$, para $\alpha<1$
	\1 \(\displaystyle M_X(t) = (1-\theta t)^{-\alpha} \), para  $t<\frac{1}{\theta}$
\end{outline}

\subsubsection{Gama($\alpha, \beta$) (\textit{shape}, \textit{rate})}
\begin{outline}
	\1 \(\displaystyle f_X(x \mid \alpha, \beta) = \frac{\beta^\alpha}{\Gamma(\alpha)} x^{\alpha-1}e^{-\beta x}\ind_{\Re_+}(x), \quad \alpha,\beta>0 \)
	\1 \(\displaystyle \E[X] = \frac{\alpha}{\beta} \)
	\1 \(\displaystyle \Var[X] = \frac{\alpha}{\beta^2} \)
	\1 Moda = $\frac{\alpha-1}{\beta}$, para $\alpha \geq 1$; $0$, para $\alpha<1$
	\1 \(\displaystyle M_X(t) = \left(1-\frac{t}{\beta}\right)^{-\alpha} \), para  $t<\beta$
\end{outline}

\subsubsection{F de Fisher-Snedecor $\fdist{n_1,n_2}$}

\begin{outline}
	\1 Sejam $U_1$ e $U_2$ variáveis aleatórias independentes tais que $U_1\sim\chidist{n_1}$ e $U_2\sim\chidist{n_2}$. Então
	\begin{equation}
		W = \frac{\dfrac{U_1}{n_1}}{\dfrac{U_2}{n_2}}
	\end{equation}
	tem distribuição F de Snedecor, com $n_1$ graus de liberdade no numerador e $n_2$ graus de liberdade no denominador.
	\1 \(\displaystyle f_W(w) = \frac{\Gamma\left(\frac{n_1+n_2}{2}\right)}{\Gamma\left(\frac{n_1}{2}\right)\Gamma\left(\frac{n_2}{2}\right)}\frac{n_1^{n_1/2}n_2^{n_2/2}w^{(n_1-2)/2}}{(n_2+wn_1)^{(n_1+n_2)/2}}, \quad w>0\)
	\1 Não existe função geradora de momentos.
	\1 $\E[W] = \dfrac{n_2}{n_2-2}$, para $n_2>2$
	\1 $\Var[W] = \dfrac{2n_2^2(n_1+n_2-2)}{n_1(n_2-2)^2(n_2-4)}$, para $n_2>4$
	\1 $\E[W^2] = \dfrac{n_2^2(n_1+2)}{n_1(n_2-2)(n_2-4)}$, para $n_2>4$
\end{outline}

\begin{theorem}[Propriedades da distribuição $F$]
	\leavevmode
	\begin{outline}[enumerate]
		\1 Se $X \sim F_{p,q}$, então $1/X \sim F_{q,p}$ (a recíproca de uma $F$ também tem distribuição $F$);
		
		\1 Se $X \sim \tdist{q}$, então $X^2 \sim F_{1,q}$;
		
		\1 Se $X \sim F_{p,q}$, então
		\[
		\frac{(p/q)X}{1 + (p/q)X} \sim \betadist{\frac{p}{2}, \frac{q}{2}}.
		\]
		
		\1 Se $V \sim \chi_r^2$ e $W \sim \chi_s^2$ independentes, então
		\[
		\frac{V/r}{W/s} \sim F_{r,s}.
		\]
	\end{outline}
\end{theorem}


\subsubsection{F de Fisher-Snedecor não central $\fdist{n_1,n_2,\lambda}$}
\begin{outline}
	\1 Sejam $U_1$ e $U_2$ variáveis aleatórias independentes tais que $U_1\sim\chidist{n_1,\lambda}$ e $U_2\sim\chidist{n_2}$. Então
	\begin{equation}
		W = \frac{\dfrac{U_1}{n_1}}{\dfrac{U_2}{n_2}}
	\end{equation}
	tem distribuição F não central, com $n_1$ graus de liberdade no numerador, $n_2$ graus de liberdade no denominador e parâmetro de não centralidade $\lambda$.
	\1 \(\displaystyle f_W(w) = \sum_{k=0}^{\infty}\frac{e^{-\lambda}\lambda^k}{k!}\frac{\Gamma\left(\frac{n_1+n_2+2k}{2}\right)}{\Gamma\left(\frac{n_1+2k}{2}\right)\Gamma\left(\frac{n_2}{2}\right)}\frac{n_1^{(n_1+2k)/2}n_2^{n_2/2}w^{(n_1+2k-2)/2}}{(n_2+wn_1)^{(n_1+n_2+2k)/2}}, \quad w>0\)
	\1 $\E[W] = \dfrac{n_2}{n_2-2}\left(1+\dfrac{2\lambda}{n_1}\right)$, para $n_2>2$
	\1 $\Var[W] = \dfrac{2n_2^2}{n_1^2(n_2-2)}\left[\dfrac{(n_1+2\lambda)^2}{(n_2-2)(n_2-4)}+\dfrac{(n_1+4\lambda)}{(n_2-4)}\right]$, para $n_2>4$
\end{outline}

\subsubsection{F não central dupla $\fdist{n_1,n_2,\lambda_1,\lambda_2}$}
\begin{outline}
	\1 Sejam $U_1$ e $U_2$ variáveis aleatórias independentes tais que $U_1\sim\chidist{n_1,\lambda_1}$ e $U_2\sim\chidist{n_2,\lambda_2}$. Então
	\begin{equation}
		W = \frac{\dfrac{U_1}{n_1}}{\dfrac{U_2}{n_2}}
	\end{equation}
	tem distribuição F não central dupla, com $n_1$ graus de liberdade no numerador, $n_2$ graus de liberdade no denominador e parâmetros de não centralidade $\lambda_1$ e $\lambda_2$.
\end{outline}

\begin{theorem}[Utilidade pública]
	Sejam $x_1, x_2, \ldots, x_n$ números reais quaisquer e $\bar{x}$ seu valor médio. Então,
	\begin{outline}[enumerate]
		\1 $\min_a \sum_{i=1}^{n} (x_i - a)^2 = \sum_{i=1}^{n} (x_i - \bar{x})^2$;
		
		\1 $\sum_{i=1}^{n} (x_i - \bar{x})^2 = \sum_{i=1}^{n} x_i^2 - n\bar{x}^2$;
		
		\1 $\sum_{i=1}^{n} (x_i - \mu)^2 = \sum_{i=1}^{n} (x_i - \bar{x})^2 + n(\bar{x} - \mu)^2$.
	\end{outline}
\end{theorem}

\section{Probabilidade e independência condicional}

\begin{theorem}[Probabilidades condicionais]
	\begin{outline}[enumerate]
		\leavevmode
		\1 \( \begin{aligned}[t]
			\P[A,B\mid C]  &= \P[A\mid B,C] \ \P[B\mid C] \\
			&= \P[B\mid A,C] \ \P[A\mid C]
		\end{aligned} \)
		\1 \( \begin{aligned}[t]
			\P[A\mid B, C]  &= \frac{\P[B\mid A,C] \ \P[A\mid C]}{\P[B\mid C]} \\
			&= \frac{\P[C\mid A,B] \ \P[A\mid B]}{P[C \mid B]}
		\end{aligned} \)
		\1 \( \begin{aligned}[t] 
			\P[A, B \mid C, D] &= \frac{\P[C | A, B, D] \P[A, B, D]}{\P[C, D]}
		\end{aligned} \)
	\end{outline}
\end{theorem}

\begin{theorem}[Independência condicional]
	\begin{outline}[enumerate]
		\leavevmode
		\1 \( \begin{aligned}[t] A  \indep B \mid C &\iff \P[A\mid B,C] = \P[A\mid C] \\ &\iff P[A, B \mid C] = \P[A\mid C] \ \P[B\mid C] \\
			&\iff [A\mid C] \indep [B\mid C]
		\end{aligned} \)
	\end{outline}
\end{theorem}


\section{Esperança, variância e covariância}

\begin{theorem}[Propriedades da esperança]
	\leavevmode
	\begin{outline}[enumerate]
		\1 \( \E[X] = \E\left[\E(X \mid Y) \right] \) (\textit{lei da esperança total})
		\1 \( \E[X \mid Z] = \E \left[ \E(X \mid Y, Z) \mid Z \right] \) (\textit{Terre})
	\end{outline}
\end{theorem}

\begin{theorem}[Propriedades da variância]
	\leavevmode
	\begin{outline}[enumerate]
		\1 \( \Var[X] = \E[(X-\mu)^2] = \E[X^2] - \mu^2\)
		\1 \( \Var[\v{X}] = \E[(\v{X}-\v{\mu}_X)(\v{X}-\v{\mu}_X)\T] = \E[\v{X}\v{X}\T] - \v{\mu}_{X}\v{\mu}_{X}\T \)
		\1 \( \Var[X] = \E(\Var[X|Y]) + \Var(\E[X|Y]) \)
		\1 \( \Var[X \mid Z] = \E(\Var[X|Y]\mid Z) + \Var(\E[X|Y] \mid Z) \)
		\1 \( \Var[X+a] = \Var[X] \)
		\1 \( \Var[aX+b] = a^2\Var[X] \)
		\1 \( \Var[\m{A}\v{X} + \v{b}] = \m{A}\Var[\v{X}]\m{A}\T \)
		\1 \( \Var[X+Y] = \Var[X]+\Var[Y]+ 2\Cov[X,Y] \)
		\1 \( \Var[\v{X}+\v{Y}] = \Var[\v{X}]+\Var[\v{Y}] + \Cov[\v{X},\v{Y}] +  \Cov[\v{Y},\v{X}] \)
		\1 \( \Var[aX+bY] = a^2\Var[X] + b^2\Var[y] + 2ab\Cov[X,Y] \)
		\1 \( \Var[aX-bY] = a^2\Var[X] + b^2\Var[y] - 2ab\Cov[X,Y] \)
		\1 \( \begin{aligned}[t] \Var\left(\sum_{i=1}^{n} X_i\right) &= \sum_{i=1}^{n} \Var(X_i) + 2\sum_{\substack{i,j=1\\i<j}}^{n} \Cov(X_i, X_j) \\
			&= \sum_{i=1}^{n} \Var(X_i) + \sum_{\substack{i,j=1\\i\neq j}}^{n} \Cov(X_i, X_j) \\
			&= \sum_{i,j=1}^{n} \Cov(X_i, X_j)
		\end{aligned} \)
		\1 \( X_1 \indep \ldots \indep X_n \implies \Var\left(\sum_{i=1}^{n} X_i\right) = \sum_{i=1}^{n} \Var(X_i) \)
		
	\end{outline}
\end{theorem}

\begin{theorem}[Propriedades da covariância]
	\leavevmode
	\begin{outline}[enumerate]
		\1 \( 
		\begin{aligned}[t]
			\Cov(X,Y) &= \E[(X-\mu_X)(Y-\mu_Y)] \\
			&= \E[XY] - \mu_X\mu_Y
		\end{aligned} \)
		\1 \(
		\begin{aligned}[t]
			\Cov(\v{X}, \v{Y}) &= \E[(\v{X} - \v{\mu}_X)(\v{Y} - \v{\mu}_Y)\T ]\\
			&= \E[\v{X}\v{Y}\T] - \v{\mu}_X\v{\mu}_Y\T
		\end{aligned} \)
		\1 \( \Cov[X,X] = \Var[X] \)
		\1 \( \Cov[\v{X},\v{X}] = \Var[\v{X}] \)
		\1 \( \Cov[X,Y] = \E[ \Cov(X,Y \mid Z)] +\Cov( \E[X\mid Z],E[Y\mid Z]) \)
		
		\1 \( \Cov(X, a) = 0 \)
		\1 \(\Cov(aX+b, cY+d) = ac\Cov(X, Y)\ \)
		\1 \(\Cov(\m{A}\v{X}+\v{b}, \m{C}\v{Y}+\v{d}) = \m{A}\Cov(\v{X}, \v{Y})\m{C}\T \)
		\1 \( \Cov(aX + bY, cW + dV) = ac\,\Cov(X, W) + ad\,\Cov(X, V) + bc\,\Cov(Y, W) + bd\,\Cov(Y, V) \)
		\1 \( \Cov(aX_1 + bX_2, cY) = ac\,\Cov(X_1, Y) + bc\,\Cov(X_2, Y) \)
		\1 \( \Cov(\m{A}\v{X}_1 + \m{B}\v{X}_2, \m{C}\v{Y}) = \m{A}\Cov(\v{X}_1, \v{Y})\m{C}\T + \m{B}\Cov(\v{X}_2, \v{Y})\m{C}\T \)
		\1 \(|\Cov(X, Y)| \leq \sqrt{\sigma^2(X)\sigma^2(Y)}\)
		
		\1 \( \rho_{X,Y} = \frac{\Cov(X, Y)}{\sigma_X \sigma_Y} \)
		\1 \( \Cov(\m{A}\v{U}, \m{B}\v{V}) = \m{A}\Cov(\v{U}, \v{V})\m{B}\T \)
		\1 \( \begin{aligned}[t] 
			\Cov(\m{A}\v{X}+ \m{B}\v{Y}, \m{C}\v{W} + \m{D}\v{V}) 
			&= \m{A}\Cov(\v{X}, \v{W})\m{C}\T \\
			&+ \m{A}\Cov(\v{X}, \v{V})\m{D}\T \\
			&+ \m{B}\Cov(\v{Y}, \v{W})\m{C}\T \\
			&+ \m{B}\Cov(\v{Y}, \v{V})\m{D}\T
		\end{aligned}\)
	\end{outline}
\end{theorem}

\begin{theorem}[Independência e covariância]
	\leavevmode
	\begin{outline}[enumerate]
		\1 $\v{X}$ e $\v{Y}$ são independentes $\implies \Cov[\v{X}, \v{Y}]=0$
		\1 Sejam $\v{X}$ e $\v{Y}$ normais 
			\2 $\v{X}$ e $\v{Y}$ são independentes $\iff \Cov[\v{X}, \v{Y}]=0$  
	\end{outline}
\end{theorem}


\section{Transformações}

\begin{theorem}[Transformação univariada contínua]
	Seja $X$ com f.d.p $f_X(x)$ e $Y=g(X)$, onde $g$ é uma função monótona. Sejam $\Xcal = \{x : f_X(x)>0\}$ e $\Ycal=\{y:y=g(x), x \in \Xcal\}$.
	Suponha que $f_X(x)$ é contínua em $\Xcal$ e $g^{-1(y)}$ tenha uma derivada continua em $y$. Então, a f.d.p. de $y$ é dada por
	\[
	f_Y(y) = f_X(g^{-1}(y))\abs{\frac{d}{dy}g^{-1}(y)}\ind_\Ycal(y).
	\]
	\begin{note}
		Caso $g$ seja monótona por partes ($n$ partes),
		\[
		f_Y(y) = \sum_{i=1}^n f_X(g_i^{-1}(y))\abs{\frac{d}{dy}g_i^{-1}(y)}\ind_\Ycal(y).
		\]
	\end{note}
\end{theorem}

\section{Vetor aleatório}

\begin{definition}
	$\v{X}$ é um \textbf{vetor aleatório} de dimensão $p\times 1$
	\begin{equation}
		\v{X} =
		\begin{bmatrix}
			X_1 \\
			X_2 \\
			\vdots \\
			X_p
		\end{bmatrix},
	\end{equation}
	em que cada $X_i$ é uma variável aleatória com \textbf{média} $\mu_i$ e \textbf{variância} $\sigma_i^2 = \sigma_{ii}$, para $i=1,\ldots,p$, em que
	\begin{align}
		\mu_i &= \E[X_i] \quad \text{(média populacional)} \\
		\sigma_i^2 &= \Var[X_i] = \E[(X_i-\mu_i)^2] \quad \text{(variância populacional)}
	\end{align}
\end{definition}

\begin{definition}
	A \textbf{covariância} e a \textbf{correlação} (populacionais) entre duas variáveis $X_i$ e $X_k$, com $i,k=1,\ldots,p$ e $i\neq k$, são dadas, respectivamente, por
	\begin{align}
		\Cov[X_i,X_k] &= \sigma_{ik} = \E[(X_i-\mu_i)(X_k-\mu_k)] \quad \text{e} \\
		\Cor(X_i,X_k) &= \rho_{ik} = \frac{\sigma_{ik}}{\sqrt{\sigma_i^2}\sqrt{\sigma_k^2}}
	\end{align}
\end{definition}

\begin{definition}
	Se $\v{X}$ é um vetor aleatório de dimensão $p\times 1$ com f.d.p. $f(\v{X})$, o \textbf{valor esperado} de uma função escalar $g(\v{X})$ é definido por
	\begin{equation}
		\E[g(\v{X})] = \int_{-\infty}^{\infty}\cdots\int_{-\infty}^{\infty} g(\v{u})f(\v{u})\,du_1\cdots du_p.
	\end{equation}
	Mais geralmente, o valor esperado de uma função vetorial de $\v{X}$, $\v{g}(\v{X})$, é dado por
	\begin{equation}
		\E[\v{g}(\v{X})] = (\E[g_i(\v{X})]),
	\end{equation}
	ou seja, o valor esperado é calculado elemento a elemento da função vetorial $\v{g}(\v{X})$.
\end{definition}

\begin{definition}[Vetor de médias e matriz de variância e covariância]
	Seja $\v{X}$ um vetor aleatório de dimensão $p\times 1$.
	O vetor de médias (populacionais) de $\v{X}$ é $\v{\mu}_{p\times 1}$ e a matriz de variâncias e covariâncias (populacionais) de $\v{X}$ é $\m{\Sigma}_{p\times p}$, em que
	\begin{equation}
		\v{\mu} = \E[\v{X}] =
		\begin{bmatrix}
			\mu_1 \\
			\mu_2 \\
			\vdots \\
			\mu_p
		\end{bmatrix}
		=
		\begin{bmatrix}
			\E[X_1] \\
			\E[X_2] \\
			\vdots \\
			\E[X_p]
		\end{bmatrix}
		\quad \text{e} \quad
		\m{\Sigma} =
		\begin{bmatrix}
			\sigma_1^2 & \sigma_{12} & \ldots & \sigma_{1p} \\
			& \sigma_2^2 & \ldots & \sigma_{2p} \\
			& & \ddots & \vdots \\
			\text{sim.} & & & \sigma_p^2
		\end{bmatrix}
	\end{equation}
	
	Matricialmente,
	\begin{equation}
		\Var[\v{X}] = \m{\Sigma} = \E[(\v{X}-\v{\mu})(\v{X}-\v{\mu})\T]
	\end{equation}
	
	\begin{note}
		$\m{\Sigma}$ é simétrica pois $\Cov[X_i,X_k] = \Cov[X_k,X_i]$, para $i,j=1,\ldots,p$.
	\end{note}
\end{definition}

\begin{definition}[Matriz de correlações]
	Seja $\v{X}$ um vetor aleatório de dimensão $p\times 1$.
	A matriz de correlações (populacionais) de $\v{X}$ é $\m{P} = \m{P}_{p\times p}$, em que
	\begin{equation}
		\m{P} = \Cor(\v{X}) =
		\begin{bmatrix}
			1 & \rho_{12} & \ldots & \rho_{1p} \\
			& 1 & \ldots & \rho_{2p} \\
			& & \ddots & \vdots \\
			\text{sim.} & & & 1
		\end{bmatrix}
	\end{equation}
	Matricialmente,
	
	\begin{equation}
		\Cor(\v{X}) = \m{P} = (\m{V}^{1/2})^{-1}\m{\Sigma}(\m{V}^{1/2})^{-1} 
		= \m{V}^{-1/2}\m{\Sigma}\m{V}^{-1/2},
	\end{equation}
		em que
	\begin{equation}
		\m{V}^{1/2} =
		\begin{bmatrix}
			\sqrt{\sigma_{11}} & 0 & \ldots & 0 \\
			0 & \sqrt{\sigma_{22}} & \ddots & 0 \\
			\vdots & \ddots & \ddots & 0 \\
			0 & \ldots & 0 & \sqrt{\sigma_{pp}}
		\end{bmatrix}
	\end{equation}
	
	\begin{note}
		\leavevmode
		\begin{outline}[enumerate]
			\1 Pelo mesmo motivo que $\m{\Sigma}$ é simétrica, $\m{P}$ também é. 
			\1 $-1\leq \rho_{ik}\leq 1$ para $i\neq k$ e $i,k=1,\ldots,p$.
		\end{outline}
	\end{note}
\end{definition}


\begin{definition}[Variância e covariância em forma matricial]
	Sejam $\v{X}$ e $\v{Y}$ vetores aleatórios $p$-dimensionais com $\E[\v{X}] = \v{\mu}_X$ e $\E[\v{Y}] = \v{\mu}_Y$. Define-se
	\begin{outline}[enumerate]
		\1 \( \Var[\v{X}] = \E[(\v{X}-\v{\mu}_X)(\v{X}-\v{\mu}_X)\T], \)
		\1 \( \Cov[\v{X},\v{Y}] = \E[(\v{X}-\v{\mu}_X)(\v{Y}-\v{\mu}_Y)\T] \)
	\end{outline}
\end{definition}

\begin{definition}[Amostra aleatória]
	Suponha que $\v{X}$ tem uma distribuição de probabilidade multivariada e que vamos observar uma amostra aleatória dessa distribuição, ou seja, $n$ observações independentes e que vêm todas da mesma distribuição:
	\begin{equation}
		\text{amostra aleatória: } \v{X}_1,\ldots,\v{X}_n,
	\end{equation}
	em que $\v{X}_i\T = (X_{i1}, X_{i2}, \ldots, X_{ip})$.
\end{definition}


\begin{definition}[Matriz aleatória]
	Suponha que $\v{X}_1,\ldots,\v{X}_n$ ainda não foram observados, mas podemos organizá-los em uma matriz aleatória
	\begin{equation}
		\m{X} =
		\begin{bmatrix}
			X_{11} & X_{12} & \ldots & X_{1p} \\
			X_{21} & X_{22} & \ldots & X_{2p} \\
			\vdots & \vdots & \ddots & \vdots \\
			X_{n1} & X_{n2} & \ldots & X_{np}
		\end{bmatrix}_{n\times p}
		=
		\begin{bmatrix}
			\v{X}_1\T \\
			\v{X}_2\T \\
			\vdots \\
			\v{X}_n\T
		\end{bmatrix}
	\end{equation}
\end{definition}

\begin{theorem}[Estimadores para $\mu$ e $\Sigma$]
	Seja $\v{X}_1,\ldots,\v{X}_n$ a.a. de uma distribuição com $\E[\v{X}] = \v{\mu}$ e $\Var[\bar{\v{X}}] = \m{\Sigma}$. Então
	\begin{outline}[enumerate]
		\1 $\bar{\v{X}}$ é um estimador não-viesado para $\v{\mu}$, com $\Var[\bar{\v{X}}] = \dfrac{\m{\Sigma}}{n}$ e
		\1 $\dfrac{n}{n-1}\m{S}$ é um estimador não-viesado para $\m{\Sigma}$, ou seja
		\begin{equation}
			\E\left[\frac{n}{n-1}\m{S}\right] = \m{\Sigma}.
		\end{equation}
	\end{outline}
	\begin{note}
		\leavevmode
		\begin{outline}[enumerate]
			\1 Os elementos de $\m{S}$ são dados por
			\begin{equation}
				s_{ik} = \frac{1}{n}\sum_{j=1}^n (x_{ji}-\bar{x}_i)(x_{jk}-\bar{x}_k)
			\end{equation}
		\end{outline}
	\end{note}
\end{theorem}
