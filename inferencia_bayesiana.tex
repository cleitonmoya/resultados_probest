\begin{theorem}[Normal com $\sigma^2$ conhecido]
	\leavevmode
	Seja $\vect{x} = (x_1, \ldots, x_n)$.
	\begin{outline}[enumerate]
		\1 Verosimilhança: \\
		\[ (X_i|\mu) \sim \normdist{\mu,\sigma^2} \] 
		
		\1 Priori:
		\[ \mu \sim \normdist{\mu_0, \sigma_0^2} \]
		
		\1 Posteriori:
		\[
		(\mu | \vect{x}) \sim \normdist{\mu_1, \sigma_1^2}
		\]
		\begin{align}
			\sigma_1^2 &= 	\frac{1}{\frac{n}{\sigma^2}+\frac{1}{\sigma_0^2}}= \frac{\sigma^2\sigma_0^2}{n\sigma_0^2+\sigma^2} \quad \\
			\mu_1 &= \sigma_1^2\left( \frac{n\bar{x}}{\sigma^2} + \frac{\mu_0}{\sigma_0^2}  \right) = 	\frac{\frac{n\bar{x}}{\sigma^2}+\frac{\mu_0}{\sigma_0^2}}{\frac{n}{\sigma^2}+\frac{1}{\sigma_0^2}}= \frac{n\bar{x}\sigma_0^2+\mu_0\sigma^2}{n\sigma_0^2+\sigma^2}
		\end{align}
		
		Fazendo $\tau=1/\sigma^2$:
		\begin{align}
			\tau_1 &= \tau_0 + n\tau \\
			\mu_1 &= \frac{n\tau\bar{x}+\tau_0\mu_0}{n\tau+\tau_0}
		\end{align}
		
		\begin{note}
			A média a posteriori é uma combinação convexa da média a priori e da média amostral.
		\end{note}
	\end{outline}
\end{theorem}

\begin{theorem}[Normal multivariada com $\matr{\Sigma}$ conhecido]
	\leavevmode
	Seja $\vect{y} = (y_1, \ldots, y_n)$.
	\begin{outline}[enumerate]
		\1 Verosimilhança:
		\[ p(\vect{y}|\vect{\mu}) = \prod_{n=1}^{N} \normdist{y_n \mid \vect{\mu}, \matr{\Sigma}} \sim \normdist{\bar{\vect{y}} \mid \vect{\mu}, \frac{1}{N}\matr{\Sigma}} \] 
		
		\1 Priori:
		\[ p(\vect{\mu}) \sim \normdist{\vect{\mu} \mid \prior{\vect{m}}, \prior{\matr{V}}} \]
		
		\1 Posteriori:
		\[
		p(\vect{\mu} | \vect{y}) \sim \normdist{\vect{\mu} \mid \post{\vect{m}}, \post{\matr{V}}}
		\]
		\begin{align}
			\post{\matr{V}}^{-1} &= \prior{\matr{V}}^{-1} + N \matr{\Sigma}^{-1} \\
			\post{\vect{m}} &= \post{\matr{V}} \left[ \matr{\Sigma}^{-1} (N\bar{\vect{y}}) + \prior{\matr{V}}^{-1} \prior{\vect{m}} \right]  
		\end{align}
	\end{outline}
\end{theorem}

\begin{theorem}[Regressão linear bayesiana com $\matr{\Sigma}$ conhecido]
	\leavevmode
	Seja $\vect{y} = (y_1, \ldots, y_n)$.
	\begin{outline}[enumerate]
		\1 Verosimilhança:
		\[ p(\vect{y}|\vect{\mu}) \sim \normdist{\bar{\vect{y}} \mid \matr{\Fcal}\vect{\mu}, \matr{\Sigma}} \] 
		
		\1 Priori:
		\[ p(\vect{\mu}) \sim \normdist{\vect{\mu} \mid \prior{\vect{m}}, \prior{\matr{V}}} \]
		
		\1 Posteriori:
		\[
		p(\vect{\mu} | \vect{y}) \sim \normdist{\vect{\mu} \mid \post{\vect{m}}, \post{\matr{V}}}
		\]
		\begin{align}
			\post{\matr{V}}^{-1} &= \prior{\matr{V}}^{-1} + \matr{\Fcal}\T \matr{\Sigma}^{-1} \matr{\Fcal} \\
			\post{\vect{m}} &= \post{\matr{V}} \left[ \matr{\Fcal}\T  \matr{\Sigma}^{-1}{\vect{y}} + \prior{\matr{V}}^{-1} \prior{\vect{m}} \right]  
		\end{align}
	\end{outline}
\end{theorem}


\begin{theorem}[Normal-Gamma]
	\leavevmode 
	\begin{outline}
		\1 Verosimilhança: 
		\[ Y_1, \ldots, Y_k \sim \normdist{\mu, \phi^{-1}}\]
		
		\1 Priori:
		\begin{align}
			(\mu,\phi) &\sim \ngdist{m, C, n, d} \\
			\mu|\phi &\sim \normdist{m, C\phi^{-1}} \\
			\phi &\sim \gammadist{n/2, /2}
		\end{align}
		
		\1 Posteriori: \[ \ (\mu,\phi)|\vect{y} \sim \ngdist{m^*, C^*, n^*, d^*}\]
		\begin{align}
			n^* &= n+k \\
			m^* &= \frac{km/C+\sum_{i=1}^{k}y_i}{k+1/C} \\
			C^* &= \frac{C}{2+kC} \\
			d^* &= d + \sum_{i=1}^{k}(y_i-\bar{y})^2 + \frac{k(\bar{y}-m)^2}{1+kC}
		\end{align}	
		
		\1 Marginais:
		\begin{align}
			\mu &\sim \tdist{n}(m,R), \quad R=C\frac{d}{n} \\
			\mu|\vect{y} &\sim \tdist{n}(m^*, R^*), \quad R^*=C^*\frac{d^*}{n^*}
		\end{align}
	\end{outline}
\end{theorem}

\begin{theorem}[Normal-Gamma Multivariada]
	\leavevmode 
	\begin{outline}
		\1 Verossimilhança: 
		\[ \vect{Y}_1, \ldots, \vect{Y}_k \mid \vect{\theta}, \phi \stackrel{iid}{\sim} \normdist{\vect{\theta}, \phi^{-1}\vect{I}_p}\]
		
		\1 Priori:
		\begin{align}
			(\vect{\theta},\phi) &\sim \ngdist{\vect{m}, \vect{C}, n, d} \\
			\vect{\theta}|\phi &\sim \normdist{\vect{m}, \vect{C}\phi^{-1}} \\
			\phi &\sim \gammadist{n/2, d/2}
		\end{align}
		
		\1 Posteriori: 
		\[ (\vect{\theta},\phi)|\vect{Y}_1, \ldots, \vect{Y}_k \sim \ngdist{\vect{m}^*, \vect{C}^*, n^*, d^*}\]
		\begin{align}
			n^* &= n+kp \\
			\vect{m}^* &= (\vect{C}^{-1} + k\vect{I}_p)^{-1}(\vect{C}^{-1}\vect{m} + k\bar{\vect{Y}}) \\
			\vect{C}^* &= (\vect{C}^{-1} + k\vect{I}_p)^{-1} \\
			d^* &= d + \sum_{j=1}^k (\vect{Y}_j - \bar{\vect{Y}})'(\vect{Y}_j - \bar{\vect{Y}}) + (\bar{\vect{Y}} - \vect{m})'(\vect{C}^{-1} + k\vect{I}_p)^{-1}(\bar{\vect{Y}} - \vect{m})
		\end{align}	
		
		\2 Caso particular $k=1$: Após observar $\vect{Y}_1 = \vect{Y}$,
		\begin{align}
			n^* &= n+p \\
			\vect{m}^* &= (\vect{C}^{-1} + \vect{I}_p)^{-1}(\vect{C}^{-1}\vect{m} + \vect{Y}) \\
			\vect{C}^* &= (\vect{C}^{-1} + \vect{I}_p)^{-1} \\
			d^* &= d + (\vect{Y} - \vect{m})'(\vect{C}^{-1} + \vect{I}_p)^{-1}(\vect{Y} - \vect{m})
		\end{align}
		
		\1 Marginais:
		\begin{align}
			\vect{\theta} &\sim \text{T}_n[\vect{m}, \vect{R}], \quad \vect{R}=\vect{C}\frac{d}{n} \\
			\vect{\theta}|\vect{Y}_1, \ldots, \vect{Y}_k &\sim \text{T}_{n^*}[\vect{m}^*, \vect{R}^*], \quad \vect{R}^*=\vect{C}^*\frac{d^*}{n^*}
		\end{align}
		
		\1 Marginais univariadas: Se $\theta_i$ é o $i$-ésimo elemento de $\vect{\theta}$, então
		\begin{align}
			\theta_i &\sim \tdist{n}(m_i, R_{ii}), \quad R_{ii}=C_{ii}\frac{d}{n} \\
			\theta_i|\vect{Y}_1, \ldots, \vect{Y}_k &\sim \tdist{n^*}(m_i^*, R_{ii}^*), \quad R_{ii}^*=C_{ii}^*\frac{d^*}{n^*}
		\end{align}
		onde $\bar{\vect{Y}} = \frac{1}{k}\sum_{j=1}^k \vect{Y}_j$
	\end{outline}
	
	\begin{note}
		\leavevmode
		\begin{outline}
			\1 O vetor de parâmetros $\vect{\theta}$ tem dimensão $p \times 1$.
			\1 Cada observação $\vect{Y}_j$ também tem dimensão $p \times 1$.
			\1 $\vect{C}$ é uma matriz $p \times p$ simétrica positiva definida.
			\1 Os graus de liberdade aumentam de $n$ para $n^* = n + kp$ (cada observação vetorial contribui com $p$ graus de liberdade).
			\1 A notação $\text{T}_n[\vect{m}, \matr{R}]$ denota a distribuição t multivariada com $n$ graus de liberdade, moda $\vect{m}$ e matriz de escala $\matr{R}$.
		\end{outline}
	\end{note}
\end{theorem}

\begin{theorem}[Normal-Gamma para Regressão Linear]
	\leavevmode 
	\begin{outline}
		\1 Verossimilhança: 
		\[ (Y_i \mid \vect{\theta}, \phi) \sim N[\vect{F}_i'\vect{\theta}, k_i\phi^{-1}], \quad i = 1, \ldots, n\]
		
		\1 Priori:
		\begin{align}
			(\vect{\theta},\phi) &\sim \text{NG}(\vect{a}, \vect{R}, \nu, s) \\
			\vect{\theta}|\phi &\sim N[\vect{a}, \vect{R}\phi^{-1}] \\
			\phi &\sim \text{Gamma}(\nu/2, s/2)
		\end{align}
		
		\1 Posteriori para múltiplas observações: 
		\[ (\vect{\theta},\phi)|Y_1, \ldots, Y_n \sim \text{NG}(\vect{a}^*, \vect{R}^*, \nu^*, s^*)\]
		\begin{align}
			\nu^* &= \nu + n \\
			Q_i &= \vect{F}_i'\vect{R}\vect{F}_i + k_i \quad \text{(variância preditiva)} \\
			e_i &= Y_i - \vect{F}_i'\vect{a} \quad \text{(erro de predição)} \\
			\vect{R}^* &= \vect{R} - \sum_{i=1}^n \frac{\vect{R}\vect{F}_i\vect{F}_i'\vect{R}}{Q_i} \\
			\vect{a}^* &= \vect{a} + \sum_{i=1}^n \frac{\vect{R}\vect{F}_i e_i}{Q_i} \\
			s^* &= s + \sum_{i=1}^n \frac{e_i^2}{Q_i}
		\end{align}	
		
		\2 Caso particular $n=1$: Após observar $(Y_1, \vect{F}_1)$ com $Y_1 = Y$, $\vect{F}_1 = \vect{F}$, $k_1 = k$,
		\begin{align}
			\nu^* &= \nu + 1 \\
			f &= \vect{F}'\vect{a} \quad \text{(predição priori)} \\
			Q &= \vect{F}'\vect{R}\vect{F} + k \quad \text{(variância preditiva priori)} \\
			\vect{R}^* &= \vect{R} - \frac{\vect{R}\vect{F}\vect{F}'\vect{R}}{Q} \\
			\vect{a}^* &= \vect{a} + \frac{\vect{R}\vect{F}(Y - f)}{Q} \\
			s^* &= s + \frac{(Y - f)^2}{Q}
		\end{align}
		
		\3 Distribuições condicionais e marginal posteriori ($n=1$):
		\begin{align}
			(\vect{\theta} \mid \phi, Y) &\sim N[\vect{a}^*, \vect{R}^*\phi^{-1}] \\
			(\phi \mid Y) &\sim \text{Gamma}\left(\frac{\nu+1}{2}, \frac{s + (Y-f)^2/Q}{2}\right)
		\end{align}
		
		\1 Marginais:
		\begin{align}
			\vect{\theta} &\sim \text{T}_\nu[\vect{a}, \vect{R}S], \quad S = s/\nu \\
			\vect{\theta}|Y_1, \ldots, Y_n &\sim \text{T}_{\nu^*}[\vect{a}^*, \vect{R}^*S^*], \quad S^* = s^*/\nu^*
		\end{align}
		
		\1 Marginais univariadas: Se $\theta_j$ é o $j$-ésimo elemento de $\vect{\theta}$, então
		\begin{align}
			\theta_j &\sim t_\nu\left(a_j, R_{jj}S\right) \\
			\theta_j|Y_1, \ldots, Y_n &\sim t_{\nu^*}\left(a_j^*, R_{jj}^*S^*\right)
		\end{align}
		ou de forma padronizada:
		\begin{align}
			\frac{\theta_j - a_j}{\sqrt{R_{jj}S}} &\sim t_\nu(0, 1) \\
			\frac{\theta_j - a_j^*}{\sqrt{R_{jj}^*S^*}} &\sim t_{\nu^*}(0, 1)
		\end{align}
		
		\1 Distribuição preditiva: Para uma nova observação $Y_{n+1}$ com covariáveis $\vect{F}_{n+1}$ e fator $k_{n+1}$,
		\begin{align}
			Y_{n+1}|Y_1, \ldots, Y_n &\sim t_{\nu^*}\left(f^*, Q^*S^*\right) \\
			f^* &= \vect{F}_{n+1}'\vect{a}^* \\
			Q^* &= \vect{F}_{n+1}'\vect{R}^*\vect{F}_{n+1} + k_{n+1}
		\end{align}
	\end{outline}
	
	\begin{note}
		\leavevmode
		\begin{outline}
			\1 O vetor de parâmetros $\vect{\theta}$ tem dimensão $p \times 1$.
			\1 Cada $Y_i$ é uma observação escalar com vetor de covariáveis $\vect{F}_i$ ($p \times 1$).
			\1 $k_i$ é o fator de escala da variância da $i$-ésima observação.
			\1 $\vect{R}$ é uma matriz $p \times p$ simétrica positiva definida.
			\1 $S = s/\nu = 1/E[\phi]$ é o inverso da precisão esperada.
			\1 Os graus de liberdade aumentam de $\nu$ para $\nu^* = \nu + n$ (número de observações escalares).
			\1 A notação $\text{T}_\nu[\vect{a}, \vect{R}S]$ denota a distribuição t multivariada com $\nu$ graus de liberdade, vetor de localização $\vect{a}$ e matriz de escala $\vect{R}S$.
			\1 A atualização sequencial é possível: a posteriori após $i$ observações serve como priori para a $(i+1)$-ésima observação.
		\end{outline}
	\end{note}
\end{theorem}
