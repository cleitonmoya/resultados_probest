\chapter*{Notação}

\begin{outline}
	\1 Variável escalar: 
		\2 Não aleatória: minúscula, itálico ($x, \theta$)
		\2 Aleatória: maiúscula, itálico ($\rv{X}, \rv{\theta}$)
			\3 Obs.: Exceto v.a. gregas (normalmente usadas em inferência Bayesiana), as quais são geralmente minúsculas.
	\1 Vetor: 
		\2 Não aleatório: minúscula, negrito, itálico ($\v{x}, \v{\theta}$)
		\2 Aleatório: maiúscula, negrito, itálico ($\v{X}, \v{\theta}$)
		\3 Obs.: Exceto v.a. gregas (normalmente usadas em inferência Bayesiana), as quais são geralmente minúsculas.
	\1 Matriz (aleatória ou não): maiúscula, negrito, reto (não itálico) ($\m{X}, \m{\Sigma}$)
\end{outline}
